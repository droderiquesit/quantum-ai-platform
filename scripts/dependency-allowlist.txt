# Every third-party package permitted in backend/Cargo.lock — direct and transitive.
# ADR 0109 (Accepted 2026-10-11): crates are admitted from this committed list,
# one package per line, `name  # ADR NNNN — reason`. A package not listed here
# fails ./scripts/check-dependencies.sh, so every admission appears in a diff.
# Transitive packages are listed too, so a crate that quietly widens its own
# dependency tree shows up here rather than nowhere.

# The decision core (ADR 0002, 0009): serde, serde_json and their closure.
serde            # ADR 0002 — serialisation of every contract
serde_json       # ADR 0002 — the wire and journal format
serde_derive     # ADR 0002 — serde's derive macro
proc-macro2      # ADR 0002 — serde_derive's proc-macro support
quote            # ADR 0002 — serde_derive's proc-macro support
syn              # ADR 0002 — serde_derive's proc-macro support
unicode-ident    # ADR 0002 — syn's identifier tables
serde_core       # ADR 0002 — serde 1.0.229 split its core traits into this crate
itoa             # ADR 0002 — serde_json integer formatting
zmij             # ADR 0002 — serde_json float formatting
memchr           # ADR 0002 — serde_json byte search

# Cold-tier compression (ADR 0113 §3, admitted under ADR 0109): qip-storage
# compresses sealed tick segments with zstd after the size measurement the ADR
# required. zstd-sys builds the reference C library, so its build closure is
# listed too; none of the build-only packages reach a shipped binary's code.
zstd             # ADR 0113 — tick-segment compression in qip-storage
zstd-safe        # ADR 0113 — zstd's safe wrapper over the C library
zstd-sys         # ADR 0113 — the reference zstd C library, built from source
cc               # ADR 0113 — build-only: compiles zstd-sys's C sources
find-msvc-tools  # ADR 0113 — build-only: cc's Windows toolchain probe
jobserver        # ADR 0113 — build-only: cc's parallel-build coordination
shlex            # ADR 0113 — build-only: cc's flag parsing
pkg-config       # ADR 0113 — build-only: zstd-sys's system-library probe
libc             # ADR 0113 — jobserver/getrandom's libc bindings
getrandom        # ADR 0113 — jobserver's dependency (build-only here)
r-efi            # ADR 0113 — getrandom's UEFI target support, never built on Linux
cfg-if           # ADR 0113 — conditional compilation helper in the closure
